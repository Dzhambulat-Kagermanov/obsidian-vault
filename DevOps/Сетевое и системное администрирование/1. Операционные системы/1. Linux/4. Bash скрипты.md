## Введение

Bash (Bourne-Again SHell) — это стандартная командная оболочка в большинстве Linux-дистрибутивов и мощный язык скриптования. Для DevOps-инженера владение Bash — это не просто умение писать однострочники, а способность автоматизировать рутинные задачи, создавать надежные скрипты для CI/CD, разворачивать инфраструктуру и диагностировать проблемы.

> [!INFO] Связь с предыдущими темами Bash-скрипты активно используют механизмы, разобранные ранее:
> 
> - [[2. Stdin, Stdout, Stderr, пайплайны, перенаправления| Потоки ввода-вывода и перенаправления]] для логирования и обработки данных
> - [[3. Ссылочные файлы (hard, soft links)| Символические ссылки]] для управления конфигурациями
> - [[2. Методология Troubleshooting| Методология troubleshooting]] для отладки скриптов


### Базовая структура скрипта

```bash
#!/bin/bash

# === Настройки и константы ===
set -euo pipefail  # строгий режим (разберем ниже)
readonly SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
readonly SCRIPT_NAME="$(basename "$0")"
readonly LOG_FILE="/var/log/${SCRIPT_NAME%.sh}.log"

# === Функции ===
log() {
    local level="$1"
    shift
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] [$level] $*" | tee -a "$LOG_FILE"
}

cleanup() {
    log "INFO" "Выполняется очистка..."
    rm -f /tmp/temp_file_*
}

# === Основная логика ===
main() {
    trap cleanup EXIT  # выполнить cleanup при выходе
    
    log "INFO" "Скрипт запущен"
    
    # ... код скрипта ...
    
    log "INFO" "Скрипт завершен успешно"
}

# Запуск main только если скрипт не source'ится
if [[ "${BASH_SOURCE[0]}" == "${0}" ]]; then
    main "$@"
fi
```

## 3. Условные конструкции

### if-elif-else

```bash
#!/bin/bash

# Базовый синтаксис
if [[ условие ]]; then
    команды
elif [[ другое_условие ]]; then
    команды
else
    команды
fi

# Примеры условий

# Сравнение строк
NAME="admin"
if [[ "$NAME" == "admin" ]]; then
    echo "Привет, админ!"
elif [[ "$NAME" == "user" ]]; then
    echo "Привет, пользователь!"
else
    echo "Неизвестный пользователь"
fi

# Операторы сравнения строк
[[ "$STR1" == "$STR2" ]]    # равны
[[ "$STR1" != "$STR2" ]]    # не равны
[[ -z "$STR" ]]             # строка пуста
[[ -n "$STR" ]]             # строка не пуста
[[ "$STR" =~ ^[0-9]+$ ]]    # регулярное выражение

# Сравнение чисел
AGE=25
if ((AGE >= 18)); then
    echo "Совершеннолетний"
fi

# Операторы сравнения чисел
((A == B))    # равно
((A != B))    # не равно
((A > B))     # больше
((A < B))     # меньше
((A >= B))    # больше или равно
((A <= B))    # меньше или равно

# Проверки файлов
FILE="/etc/nginx/nginx.conf"
if [[ -f "$FILE" ]]; then
    echo "Файл существует"
fi

# Операторы проверки файлов
[[ -e "$PATH" ]]    # существует (файл или директория)
[[ -f "$PATH" ]]    # обычный файл
[[ -d "$PATH" ]]    # директория
[[ -L "$PATH" ]]    # символическая ссылка
[[ -r "$PATH" ]]    # читаемый
[[ -w "$PATH" ]]    # записываемый
[[ -x "$PATH" ]]    # исполняемый
[[ -s "$PATH" ]]    # размер > 0
[[ "$FILE1" -nt "$FILE2" ]]  # FILE1 новее FILE2
[[ "$FILE1" -ot "$FILE2" ]]  # FILE1 старше FILE2

# Логические операторы
if [[ -f "$FILE" && -r "$FILE" ]]; then
    echo "Файл существует и читаемый"
fi

if [[ "$AGE" -ge 18 || "$HAS_PARENT" == "true" ]]; then
    echo "Доступ разрешен"
fi

if [[ ! -d "$DIR" ]]; then
    echo "Директория не существует"
fi
```

### Case (выбор из множества вариантов)

```bash
#!/bin/bash

ACTION="$1"

case "$ACTION" in
    start)
        echo "Запуск сервиса..."
        systemctl start myapp
        ;;
    stop)
        echo "Остановка сервиса..."
        systemctl stop myapp
        ;;
    restart|reload)
        echo "Перезапуск сервиса..."
        systemctl restart myapp
        ;;
    status)
        systemctl status myapp
        ;;
    *)
        echo "Использование: $0 {start|stop|restart|status}"
        exit 1
        ;;
esac

# Паттерны в case
FILENAME="backup-2024-01-15.tar.gz"

case "$FILENAME" in
    *.tar.gz|*.tgz)
        echo "Архив tar.gz"
        tar -tzf "$FILENAME" | head -5
        ;;
    *.zip)
        echo "Архив zip"
        unzip -l "$FILENAME" | head -5
        ;;
    *.log)
        echo "Лог-файл"
        tail -20 "$FILENAME"
        ;;
    backup-*)
        echo "Файл бэкапа"
        ;;
    *)
        echo "Неизвестный тип файла"
        ;;
esac
```

### While loop

```bash
#!/bin/bash

# Базовый while
COUNTER=0
while ((COUNTER < 5)); do
    echo "Counter: $COUNTER"
    ((COUNTER++))
done

# While с чтением файла построчно
while IFS= read -r LINE; do
    echo "Line: $LINE"

while ! ping -c 1 google.com &>/dev/null; do
    echo "Ожидание сети..."
    sleep 5
done

# Бесконечный цикл (для демонов/мониторинга)
while true; do
    check_service_status
    sleep 60
done

# While с условием выхода
while read -r -p "Введите имя (или 'exit'): " NAME; do
    if [[ "$NAME" == "exit" ]]; then
        break
    fi
    echo "Привет, $NAME!"
done
```

## 6. Обработка ошибок и отладка

### Строгий режим: set -euo pipefail

```bash
#!/bin/bash
set -euo pipefail

# -e: выход при любой ошибке (non-zero exit code)
# -u: ошибка при использовании необъявленной переменной
# -o pipefail: pipe возвращает код ошибки последней упавшей команды

# Примеры:
set -e
false  # скрипт немедленно завершится

set -u
echo "$UNDEFINED_VAR"  # ошибка: unbound variable

set -o pipefail
false | true  # вернет код 1 (от false), а не 0
```

### Trap: обработка сигналов и ошибок

```bash
#!/bin/bash

# Trap для очистки при выходе
cleanup() {
    echo "Выполняется очистка..."
    rm -f /tmp/temp_*
    # Отмена блокировки, закрытие соединений и т.д.
}
trap cleanup EXIT

# Trap для обработки ошибок
error_handler() {
    local exit_code=$?
    local line_number=$1
    echo "Ошибка в строке $line_number, код выхода: $exit_code" >&2
    # Логирование, уведомление и т.д.
}
trap 'error_handler $LINENO' ERR

# Trap для сигналов
trap 'echo "Получен SIGINT, завершение..."; exit 130' INT
trap 'echo "Получен SIGTERM, завершение..."; exit 143' TERM

# Основная логика
echo "Работа скрипта..."
sleep 10
```

### Проверка кодов возврата

```bash
#!/bin/bash

# Базовая проверка
if command; then
    echo "Успех"
else
    echo "Ошибка"
fi

# Сохранение кода возврата
./risky_command
EXIT_CODE=$?
if ((EXIT_CODE != 0)); then
    echo "Команда завершилась с кодом $EXIT_CODE" >&2
    exit $EXIT_CODE
fi

# Игнорирование ошибок (если ожидаемы)
./may_fail || true
./may_fail || echo "Предупреждение: команда не выполнена"

# Retry логика
retry_command() {
    local cmd="$1"
    local max_attempts=3
    local attempt=1
    
    while ((attempt <= max_attempts)); do
        echo "Попытка $attempt из $max_attempts..."
        if eval "$cmd"; then
            return 0
        fi
        ((attempt++))
        sleep 2
    done
    
    echo "Все попытки исчерпаны" >&2
    return 1
}

retry_command "curl -f https://api.example.com/health"
```

### Работа с датами и временем

```bash
#!/bin/bash

# Текущая дата в разных форматах
echo $(date +%F)           # 2026-01-15
echo $(date +%T)           # 14:30:45
echo $(date '+%Y-%m-%d_%H-%M-%S')  # 2026-01-15_14-30-45
echo $(date +%s)           # Unix timestamp

# Дата N дней назад/вперед
PAST_DATE=$(date -d "7 days ago" +%F)
FUTURE_DATE=$(date -d "+30 days" +%F)

# Разница между датами в секундах
DATE1=$(date -d "2026-01-01" +%s)
DATE2=$(date -d "2026-01-15" +%s)
DIFF_SECONDS=$((DATE2 - DATE1))
DIFF_DAYS=$((DIFF_SECONDS / 86400))
echo "Разница: $DIFF_DAYS дней"

# Проверка, старше ли файл N дней
if [[ $(find /var/log/app.log -mtime +7) ]]; then
    echo "Файл старше 7 дней"
```

### Избегайте антипаттернов

```bash
#!/bin/bash

# ❌ ПЛОХО: использование cat в pipe (Useless Use of Cat)
cat file.txt | grep "pattern"

# ✅ ХОРОШО: grep читает файл напрямую
grep "pattern" file.txt

# ❌ ПЛОХО: цикл while с pipe (создает подшелл)
cat file.txt | while read -r line; do
    COUNTER=$((COUNTER + 1))  # COUNTER не изменится вне цикла!
done
echo "$COUNTER"  # 0

# ✅ ХОРОШО: перенаправление ввода
while read -r line; do
    ((COUNTER++))
done < file.txt
echo "$COUNTER"  # правильное значение

# ❌ ПЛОХО: eval с пользовательским вводом
USER_INPUT="rm -rf /"
eval "$USER_INPUT"  # КАТАСТРОФА!

# ✅ ХОРОШО: избегайте eval или строго валидируйте
if [[ "$USER_INPUT" =~ ^[a-zA-Z0-9_]+$ ]]; then
    eval "$USER_INPUT"
fi

# ❌ ПЛОХО: жесткие пути
cd /opt/myapp/current

# ✅ ХОРОШО: относительные пути или переменные
SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
cd "$SCRIPT_DIR/../current"
```

### Сценарий 1: Мониторинг дискового пространства с алертами

```bash
#!/bin/bash
set -euo pipefail

THRESHOLD=80
ALERT_EMAIL="ops@example.com"

check_disk_usage() {
    local mount_point="$1"
    local usage=$(df -h "$mount_point" | awk 'NR==2 {print $5}' | tr -d '%')
    
    if ((usage > THRESHOLD)); then
        echo "WARNING: $mount_point использует ${usage}% места" >&2
        echo "Диск $mount_point на сервере $(hostname) заполнен на ${usage}%" | \
            mail -s "Alert: Disk Usage" "$ALERT_EMAIL"
        return 1
    fi
    
    echo "OK: $mount_point использует ${usage}%"
    return 0
}

# Проверка всех точек монтирования
ERRORS=0
while read -r mount_point; do
    if ! check_disk_usage "$mount_point"; then
        ((ERRORS++))
    fi
done < <(df -h | awk 'NR>1 && $1 ~ /^\/dev/ {print $6}')

if ((ERRORS > 0)); then
    echo "Обнаружено $ERRORS проблем" >&2
    exit 1
fi

echo "Все диски в норме"
```

### Сценарий 2: Ротация логов с архивацией

```bash
#!/bin/bash
set -euo pipefail

LOG_DIR="/var/log/myapp"
RETENTION_DAYS=30
ARCHIVE_DIR="/backup/logs"

rotate_logs() {
    local log_file="$1"
    local base_name=$(basename "$log_file")
    local timestamp=$(date +%Y%m%d_%H%M%S)
    
    # Сжатие и перемещение
    gzip -c "$log_file" > "$ARCHIVE_DIR/${base_name}_${timestamp}.gz"
    
    # Очистка оригинала (сохраняем дескриптор для работающих процессов)
    truncate -s 0 "$log_file"
    
    echo "Ротация: $log_file → ${base_name}_${timestamp}.gz"
}

# Удаление старых архивов
find "$ARCHIVE_DIR" -name "*.gz" -mtime +$RETENTION_DAYS -delete

# Ротация активных логов
for log_file in "$LOG_DIR"/*.log; do
    if [[ -f "$log_file" && -s "$log_file" ]]; then
        rotate_logs "$log_file"
    fi
done

echo "Ротация завершена"
```

### Сценарий 3: Health check для микросервисов

```bash
#!/bin/bash
set -euo pipefail

SERVICES=(
    "api-gateway:http://localhost:8080/health"
    "user-service:http://localhost:8081/health"
    "order-service:http://localhost:8082/health"
)

TIMEOUT=5
FAILED=()

check_service() {
    local name="$1"
    local url="$2"
    
    if curl -sf --max-time "$TIMEOUT" "$url" > /dev/null; then
        echo "✅ $name: OK"
        return 0
    else
        echo "❌ $name: FAILED"
        return 1
    fi
}

echo "=== Health Check $(date) ==="

for service in "${SERVICES[@]}"; do
    IFS=':' read -r name protocol rest <<< "$service"
    url="${protocol}:${rest}"
    
    if ! check_service "$name" "$url"; then
        FAILED+=("$name")
    fi
done

echo ""
if (( ${#FAILED[@]} > 0 )); then
    echo "⚠️  Нездоровые сервисы: ${FAILED[*]}"
    exit 1
else
    echo "✅ Все сервисы работают"
fi
```
